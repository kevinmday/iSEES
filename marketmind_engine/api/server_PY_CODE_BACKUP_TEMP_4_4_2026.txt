engine_logs = deque(maxlen=1000)

def log(message: str):
    entry = f"{time.strftime('%H:%M:%S')} | [{MM_ROLE}] {message}"
    print(entry)
    engine_logs.append(entry)

class CapturePrint:
    def __init__(self, original):
        self.original = original

    def write(self, data):
        self.original.write(data)
        if "[STATE]" in data:
            engine_logs.append(data.strip())

    def flush(self):
        self.original.flush()

sys.stdout = CapturePrint(sys.stdout)

def normalize_event(e):
    try:
        return (
            getattr(e, "title", "").strip(),
            getattr(e, "source", "").strip(),
            str(getattr(e, "timestamp", ""))[:16],
        )
    except Exception:
        return str(e)

def is_market_open():
    try:
        now = datetime.now(pytz.timezone("US/Eastern"))
        if now.weekday() >= 5:
            return False
        return now.replace(hour=9, minute=30) <= now <= now.replace(hour=16, minute=0)
    except Exception:
        return True

def safe_propagation_update(engine, symbols):
    log("[DEBUG] entering safe_propagation_update")
    try:
        if hasattr(engine, "update"):
            engine.update(symbols)
    except Exception as e:
        log(f"[PROP ERROR] {e}")

def generate_snapshot():

    global latest_events

    rows_map = {}

    for line in list(engine_logs)[-300:]:

        if "[STATE]" in line:
            try:
                parts = line.split()
                symbol = parts[1]

                drift = 0.0
                ucip = 0.0
                life = 0

                for p in parts:
                    if p.startswith("DRIFT="):
                        drift = float(p.split("=")[1])
                    elif p.startswith("UCIP="):
                        ucip = float(p.split("=")[1])
                    elif p.startswith("LIFE="):
                        life = int(p.split("=")[1])

                if life == 0:
                    continue

                score = drift * ucip * life

                rows_map[symbol] = {
                    "symbol": symbol,
                    "score": score,
                    "drift": drift,
                    "ucip": ucip,
                    "life": life,
                }

            except Exception:
                pass

    rows = list(rows_map.values())

    if not rows:
        log("[SNAPSHOT] no data")
        return

    signal_results = signal_layer.compute(rows)
    signal_results = sorted(signal_results, key=lambda x: x.get("score", 0), reverse=True)

    candidates = execution_layer.evaluate(signal_results)

    top = signal_results[:5]
    avg = sum([r.get("drift", 0) for r in signal_results]) / len(signal_results)

    log("================ SNAPSHOT ================")
    log(f"TIME: {time.strftime('%H:%M:%S')}")
    log(f"SYMBOLS: {len(signal_results)}")

    log("TOP SIGNALS:")
    for r in top:
        log(
            f"  {r.get('symbol','NA'):<6} "
            f"score={r.get('score',0):+.4f}  "
            f"Δ={r.get('delta',0):+.4f}  "
            f"ΔΔ={r.get('accel',0):+.4f}  "
            f"p={r.get('persistence', r.get('persist', 1)):<2}  "
            f"conf={r.get('confidence',0):.4f} "
            f"→ {r.get('confidence_label','NA'):<6} | {r.get('signal','NA')}"
        )

    log(f"AVG DRIFT: {avg:+.6f}")
    log("========================================")

    execution_layer.print_candidates(candidates)

    # WHY LAYER
    if latest_events:
        why_results = []

        for r in top:
            symbol = r.get("symbol")
            if symbol:
                why_results.append(
                    why_layer.analyze(symbol, latest_events)
                )

        why_layer.print_why(why_results)

def engine_loop():

    global engine_loop_running
    global last_rss_poll, last_benzinga_poll
    global evaluation_symbols, latest_events

    while engine_loop_running:

        try:

            if engine_controller.is_running():

                now = time.time()
                symbols = set()

                rss_events = []
                benzinga_events = []

                # -------------------------
                # RSS
                # -------------------------
                if rss_service and (now - last_rss_poll > RSS_POLL_INTERVAL):

                    rss_service.worker.poll_once()
                    rss_service._update_projection()

                    last_rss_poll = now
                    rss_events = rss_service.get_projection_events()

                    log(f"RSS events: {len(rss_events)}")

                    if rss_events:
                        log(f"[DEBUG_EVENT_0] {normalize_event(rss_events[0])}")

                # -------------------------
                # BENZINGA 🔥 NEW
                # -------------------------
                if now - last_benzinga_poll > BENZINGA_POLL_INTERVAL:

                    benzinga_events = fetch_benzinga_news()
                    last_benzinga_poll = now

                    if benzinga_events:
                        log(f"[BENZINGA] events: {len(benzinga_events)}")

                # -------------------------
                # MERGE EVENTS
                # -------------------------
                latest_events = (rss_events or []) + (benzinga_events or [])

                # -------------------------
                # SYMBOL EXTRACTION
                # -------------------------
                for e in latest_events:

                    if hasattr(e, "symbol") and e.symbol:
                        symbols.add(e.symbol)

                    if hasattr(e, "symbols"):
                        symbols.update(e.symbols)

                    # benzinga dict format
                    if isinstance(e, dict):
                        for s in e.get("symbols", []):
                            symbols.add(s)

                validated = [s for s in symbols if symbol_validator.is_valid(s)]
                evaluation_symbols = validated[:30]

                log(f"Symbols: {evaluation_symbols}")

                # -------------------------
                # RUN ENGINE
                # -------------------------
                if is_market_open():
                    for s in evaluation_symbols:
                        engine_controller.run_symbol_cycle(s)

                safe_propagation_update(propagation_engine, evaluation_symbols)

                generate_snapshot()

        except Exception as e:
            log(f"Engine loop error: {e}")

        time.sleep(2)

@app.post("/api/engine/start", response_model=StartResponse)
def start_engine():
    global engine_loop_running, engine_loop_thread

    engine_controller.start()

    if not engine_loop_running:
        engine_loop_running = True
        engine_loop_thread = threading.Thread(target=engine_loop, daemon=True)
        engine_loop_thread.start()

    log("Engine started")
    return {"status": "started"}
